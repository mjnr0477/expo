import React, { useState, useRef } from "react";
import {
  SafeAreaView,
  View,
  Text,
  TouchableOpacity,
  FlatList,
  Image,
  TextInput,
  Modal,
  StyleSheet,
  KeyboardAvoidingView,
  Platform,
  StatusBar,
  Alert,
} from "react-native";

/*
  Pulse - Stage 1 (Single-file prototype)
  - Bottom tabs: Home / Chat / Create / Trusted / Profile
  - Local state only (works immediately in Snack + Expo Go)
  - Add posts, like, chat (local), add trusted users
*/

const SAMPLE_POSTS = [
  {
    id: "1",
    author: "Amina",
    avatar: "https://i.pravatar.cc/150?img=12",
    text: "Feeling alive today ✨",
    image: "https://picsum.photos/600/400?random=1",
    likes: 3,
  },
  {
    id: "2",
    author: "Daniel",
    avatar: "https://i.pravatar.cc/150?img=5",
    text: "Morning run gave me clarity.",
    image: "https://picsum.photos/600/400?random=2",
    likes: 7,
  },
];

const SAMPLE_CHATS = [
  {
    id: "c1",
    name: "Amina",
    last: "See you later!",
    messages: [
      { id: "m1", fromMe: false, text: "Hey are you free tonight?" },
      { id: "m2", fromMe: true, text: "Yes — let's meet at 7." },
    ],
  },
  {
    id: "c2",
    name: "Daniel",
    last: "Nice!",
    messages: [{ id: "m1", fromMe: false, text: "Have you seen the new track?" }],
  },
];

export default function App() {
  const [tab, setTab] = useState("Home");
  const [posts, setPosts] = useState(SAMPLE_POSTS);
  const [chats, setChats] = useState(SAMPLE_CHATS);
  const [selectedChat, setSelectedChat] = useState(null);
  const [trusted, setTrusted] = useState(["Amina"]);
  const [createModal, setCreateModal] = useState(false);

  // Create post fields
  const [postText, setPostText] = useState("");
  const [postImage, setPostImage] = useState("");

  // Chat composer
  const [chatText, setChatText] = useState("");
  const chatScrollRef = useRef(null);

  // Profile
  const [username, setUsername] = useState("You");

  // Helpers
  function toggleLike(postId) {
    setPosts((prev) =>
      prev.map((p) => (p.id === postId ? { ...p, likes: p.likes + 1 } : p))
    );
  }

  function addPost() {
    if (!postText.trim() && !postImage.trim()) {
      Alert.alert("Empty post", "Write something or add an image URL.");
      return;
    }
    const newPost = {
      id: Date.now().toString(),
      author: username,
      avatar: `https://i.pravatar.cc/150?u=${username}`,
      text: postText,
      image: postImage || `https://picsum.photos/600/400?random=${Date.now() % 100}`,
      likes: 0,
    };
    setPosts((p) => [newPost, ...p]);
    setPostText("");
    setPostImage("");
    setCreateModal(false);
    setTab("Home");
  }

  function openChat(chatId) {
    const c = chats.find((x) => x.id === chatId);
    setSelectedChat(c || null);
    setTab("ChatSingle");
  }

  function sendMessage() {
    if (!chatText.trim()) return;
    if (!selectedChat) return;
    const newMsg = { id: Date.now().toString(), fromMe: true, text: chatText };
    setChats((prev) =>
      prev.map((c) =>
        c.id === selectedChat.id ? { ...c, messages: [...c.messages, newMsg], last: newMsg.text } : c
      )
    );
    // update selectedChat locally too
    setSelectedChat((prev) => ({ ...prev, messages: [...prev.messages, newMsg] }));
    setChatText("");
    // Small timeout to scroll (UI only)
    setTimeout(() => {
      try {
        chatScrollRef.current?.scrollToEnd({ animated: true });
      } catch (e) {}
    }, 100);
  }

  function addTrusted(name) {
    if (!name.trim()) return;
    if (trusted.includes(name)) {
      Alert.alert("Already trusted", `${name} is already in your trusted circle.`);
      return;
    }
    setTrusted((t) => [name, ...t]);
  }

  function removeTrusted(name) {
    setTrusted((t) => t.filter((x) => x !== name));
  }

  // UI components
  const Header = ({ title }) => (
    <View style={styles.header}>
      <Text style={styles.headerTitle}>{title}</Text>
      <Text style={styles.headerSub}>Pulse — feel, connect, create</Text>
    </View>
  );

  const FeedItem = ({ item }) => (
    <View style={styles.post}>
      <View style={styles.postHeader}>
        <Image source={{ uri: item.avatar }} style={styles.avatar} />
        <View style={{ flex: 1 }}>
          <Text style={styles.postAuthor}>{item.author}</Text>
          <Text style={styles.postTextSmall}>{new Date(parseInt(item.id, 10) || Date.now()).toLocaleString()}</Text>
        </View>
        <TouchableOpacity onPress={() => toggleLike(item.id)} style={styles.likeBtn}>
          <Text>❤️ {item.likes}</Text>
        </TouchableOpacity>
      </View>
      <Text style={styles.postBodyText}>{item.text}</Text>
      {item.image ? (
        <Image source={{ uri: item.image }} style={styles.postImage} resizeMode="cover" />
      ) : null}
    </View>
  );

  const Home = () => (
    <SafeAreaView style={styles.container}>
      <Header title="Home" />
      <FlatList
        data={posts}
        keyExtractor={(i) => i.id}
        renderItem={FeedItem}
        contentContainerStyle={{ padding: 12, paddingBottom: 120 }}
        ListEmptyComponent={<Text style={{ padding: 20 }}>No posts yet — create one!</Text>}
      />
    </SafeAreaView>
  );

  const ChatList = () => (
    <SafeAreaView style={styles.container}>
      <Header title="Chats" />
      <FlatList
        data={chats}
        keyExtractor={(i) => i.id}
        renderItem={({ item }) => (
          <TouchableOpacity style={styles.chatRow} onPress={() => openChat(item.id)}>
            <Image source={{ uri: `https://i.pravatar.cc/150?u=${item.name}` }} style={styles.avatarSmall} />
            <View style={{ flex: 1 }}>
              <Text style={styles.chatName}>{item.name}</Text>
              <Text style={styles.chatLast}>{item.last}</Text>
            </View>
          </TouchableOpacity>
        )}
        ItemSeparatorComponent={() => <View style={{ height: 1, backgroundColor: "#eee", marginHorizontal: 12 }} />}
      />
    </SafeAreaView>
  );

  const ChatSingle = () => {
    if (!selectedChat) return <ChatList />;
    return (
      <SafeAreaView style={styles.container}>
        <Header title={selectedChat.name} />
        <View style={{ flex: 1, padding: 12 }}>
          <FlatList
            ref={chatScrollRef}
            data={selectedChat.messages}
            keyExtractor={(m) => m.id}
            renderItem={({ item }) => (
              <View style={[styles.chatBubble, item.fromMe ? styles.chatFromMe : styles.chatFromThem]}>
                <Text style={{ color: item.fromMe ? "#fff" : "#000" }}>{item.text}</Text>
              </View>
            )}
            contentContainerStyle={{ paddingBottom: 20 }}
          />
        </View>
        <KeyboardAvoidingView behavior={Platform.OS === "ios" ? "padding" : undefined}>
          <View style={styles.chatComposer}>
            <TextInput
              placeholder="Write a message..."
              style={styles.chatInput}
              value={chatText}
              onChangeText={setChatText}
            />
            <TouchableOpacity style={styles.sendBtn} onPress={sendMessage}>
              <Text style={{ color: "#fff" }}>Send</Text>
            </TouchableOpacity>
          </View>
        </KeyboardAvoidingView>
      </SafeAreaView>
    );
  };

  const Trusted = () => {
    const [newName, setNewName] = useState("");
    return (
      <SafeAreaView style={styles.container}>
        <Header title="Trusted" />
        <View style={{ padding: 12 }}>
          <Text style={{ marginBottom: 6 }}>Add trusted person (name)</Text>
          <View style={{ flexDirection: "row" }}>
            <TextInput
              style={[styles.input, { flex: 1 }]}
              placeholder="e.g. Amina"
              value={newName}
              onChangeText={setNewName}
            />
            <TouchableOpacity
              onPress={() => {
                addTrusted(newName.trim());
                setNewName("");
              }}
              style={[styles.primaryBtn, { marginLeft: 8, alignSelf: "center" }]}
            >
              <Text style={{ color: "#fff" }}>Add</Text>
            </TouchableOpacity>
          </View>

          <Text style={{ marginTop: 18, marginBottom: 6 }}>Your trusted circle</Text>
          {trusted.length === 0 ? (
            <Text>No one yet.</Text>
          ) : (
            trusted.map((t) => (
              <View key={t} style={styles.trustedRow}>
                <Text>{t}</Text>
                <TouchableOpacity onPress={() => removeTrusted(t)}>
                  <Text style={{ color: "#c00" }}>Remove</Text>
                </TouchableOpacity>
              </View>
            ))
          )}
        </View>
      </SafeAreaView>
    );
  };

  const Profile = () => (
    <SafeAreaView style={styles.container}>
      <Header title="Profile" />
      <View style={{ padding: 20 }}>
        <Image source={{ uri: `https://i.pravatar.cc/150?u=${username}` }} style={styles.bigAvatar} />
        <Text style={{ fontSize: 20, fontWeight: "700", marginTop: 12 }}>{username}</Text>
        <Text style={{ marginTop: 6, color: "#666" }}>Pulse member</Text>

        <View style={{ marginTop: 20 }}>
          <Text>Edit display name</Text>
          <TextInput style={styles.input} value={username} onChangeText={setUsername} />
        </View>

        <View style={{ marginTop: 20 }}>
          <TouchableOpacity
            style={styles.primaryBtn}
            onPress={() => Alert.alert("Profile saved", "Your display name was updated.")}
          >
            <Text style={{ color: "#fff" }}>Save Profile</Text>
          </TouchableOpacity>
        </View>
      </View>
    </SafeAreaView>
  );

  // Main render
  return (
    <View style={{ flex: 1, paddingTop: StatusBar.currentHeight || 0 }}>
      {tab === "Home" && <Home />}
      {tab === "Chat" && <ChatList />}
      {tab === "ChatSingle" && <ChatSingle />}
      {tab === "Trusted" && <Trusted />}
      {tab === "Profile" && <Profile />}

      <Modal visible={createModal} animationType="slide" onRequestClose={() => setCreateModal(false)}>
        <SafeAreaView style={{ flex: 1 }}>
          <View style={{ padding: 16 }}>
            <Text style={{ fontSize: 18, fontWeight: "700" }}>Create Post</Text>
            <TextInput
              placeholder="What's on your pulse?"
              value={postText}
              onChangeText={setPostText}
              multiline
              style={[styles.input, { height: 100, marginTop: 12 }]}
            />
            <TextInput
              placeholder="Optional image URL"
              value={postImage}
              onChangeText={setPostImage}
              style={[styles.input, { marginTop: 12 }]}
            />
            <View style={{ flexDirection: "row", marginTop: 16 }}>
              <TouchableOpacity style={[styles.primaryBtn, { flex: 1 }]} onPress={addPost}>
                <Text style={{ color: "#fff" }}>Post</Text>
              </TouchableOpacity>
              <TouchableOpacity
                style={[styles.secondaryBtn, { flex: 1, marginLeft: 8 }]}
                onPress={() => setCreateModal(false)}
              >
                <Text>Cancel</Text>
              </TouchableOpacity>
            </View>
          </View>
        </SafeAreaView>
      </Modal>

      {/* Bottom Tab */}
      <View style={styles.bottom}>
        <TouchableOpacity style={styles.tabBtn} onPress={() => setTab("Home")}>
          <Text style={tab === "Home" ? styles.tabActive : styles.tabText}>Home</Text>
        </TouchableOpacity>

        <TouchableOpacity style={styles.tabBtn} onPress={() => setTab("Chat")}>
          <Text style={tab === "Chat" || tab === "ChatSingle" ? styles.tabActive : styles.tabText}>Chat</Text>
        </TouchableOpacity>

        <TouchableOpacity
          style={[styles.tabBtn, { backgroundColor: "#111", borderRadius: 30, paddingHorizontal: 16 }]}
          onPress={() => setCreateModal(true)}
        >
          <Text style={{ color: "#fff", fontWeight: "700" }}>Create</Text>
        </TouchableOpacity>

        <TouchableOpacity style={styles.tabBtn} onPress={() => setTab("Trusted")}>
          <Text style={tab === "Trusted" ? styles.tabActive : styles.tabText}>Trusted</Text>
        </TouchableOpacity>

        <TouchableOpacity style={styles.tabBtn} onPress={() => setTab("Profile")}>
          <Text style={tab === "Profile" ? styles.tabActive : styles.tabText}>Profile</Text>
        </TouchableOpacity>
      </View>
    </View>
  );
}

const styles = StyleSheet.create({
  header: { padding: 14, backgroundColor: "#fff", borderBottomWidth: 1, borderColor: "#eee" },
  headerTitle: { fontSize: 20, fontWeight: "700" },
  headerSub: { color: "#666", marginTop: 4, fontSize: 12 },

  container: { flex: 1, backgroundColor: "#fafafa" },

  post: { backgroundColor: "#fff", marginBottom: 12, borderRadius: 8, overflow: "hidden", paddingBottom: 8 },
  postHeader: { flexDirection: "row", alignItems: "center", padding: 12 },
  avatar: { width: 48, height: 48, borderRadius: 24, marginRight: 12 },
  avatarSmall: { width: 44, height: 44, borderRadius: 22, marginRight: 12, margin: 12 },
  postAuthor: { fontWeight: "700" },
  postTextSmall: { fontSize: 11, color: "#888" },
  postBodyText: { paddingHorizontal: 12, paddingBottom: 8 },
  postImage: { width: "100%", height: 200 },

  likeBtn: { padding: 8 },

  chatRow: { flexDirection: "row", alignItems: "center", padding: 12 },
  chatName: { fontWeight: "700" },
  chatLast: { color: "#666", marginTop: 4 },

  chatBubble: { marginVertical: 6, padding: 10, maxWidth: "80%", borderRadius: 14 },
  chatFromMe: { backgroundColor: "#111", alignSelf: "flex-end" },
  chatFromThem: { backgroundColor: "#eee", alignSelf: "flex-start" },

  chatComposer: { flexDirection: "row", padding: 8, borderTopWidth: 1, borderColor: "#eee", alignItems: "center" },
  chatInput: { flex: 1, padding: 10, backgroundColor: "#fff", borderRadius: 8, borderWidth: 1, borderColor: "#eee" },
  sendBtn: { backgroundColor: "#111", paddingHorizontal: 12, paddingVertical: 10, borderRadius: 8, marginLeft: 8 },

  input: { borderWidth: 1, borderColor: "#ddd", padding: 10, borderRadius: 6, backgroundColor: "#fff" },

  trustedRow: { flexDirection: "row", justifyContent: "space-between", paddingVertical: 8 },

  bigAvatar: { width: 110, height: 110, borderRadius: 55 },

  primaryBtn: { backgroundColor: "#111", padding: 12, borderRadius: 8, alignItems: "center" },

  bottom: {

    backgroundColor: "#fff",

    borderColor: "#eee",
    flexDirection: "row",
    alignItems: "center",
    justifyContent: "space-around",
    paddingHorizontal: 6,
  },
  tabBtn: { padding: 8 },
  tabText: { color: "#666" },
  tabActive: { color: "#000", fontWeight: "700" },
});
